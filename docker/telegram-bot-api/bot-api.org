** Docker contentns

FROM alpine:latest AS build-env
ENV CXXFLAGS=""
WORKDIR /src
RUN apk add --no-cache alpine-sdk linux-headers git zlib-dev openssl-dev gperf cmake
COPY CMakeLists.txt .clang-format ./
COPY .git .git
COPY td td
COPY telegram-bot-api telegram-bot-api
RUN rm -rf build & mkdir build
WORKDIR /src/build
RUN cmake -DCMAKE_BUILD_TYPE=Release .. -DCMAKE_INSTALL_PREFIX:PATH=.. ..
RUN echo building with $(nproc) cores && export CXXFLAGS="" && cmake --build . -j $(nproc) --target install
RUN ls -R /src

FROM nginx:1.19-alpine AS exec-env
WORKDIR /app
RUN apk add --no-cache openssl zlib libstdc++
COPY --from=build-env /src/bin/telegram-bot-api .
RUN mkdir /file && chown nginx:nginx /file -R && chmod ugo+rw /file -R
RUN adduser -S botapi -G nginx
COPY nginx.conf /etc/nginx/nginx.conf
COPY start_api.sh .
RUN chmod +x start_api.sh
ENTRYPOINT [ "./start_api.sh"]
